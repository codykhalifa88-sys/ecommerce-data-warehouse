# apache-airflow is installed together with everything below against its
# official constraints file (see Dockerfile / README / CI), e.g.:
#   pip install -r requirements.txt \
#     --constraint "https://raw.githubusercontent.com/apache/airflow/constraints-2.9.1/constraints-3.11.txt"
#
# Packages Airflow itself depends on (pandas, numpy, psycopg2-binary,
# python-dotenv, pytest, PyYAML, alembic) are intentionally left UNPINNED
# here — the constraints file already pins them to versions known to work
# with apache-airflow==2.9.1, and adding our own conflicting pin (e.g.
# pandas==2.2.2 vs. the constraint's pandas==2.1.4) breaks `pip install`
# with a ResolutionImpossible error. Packages Airflow doesn't touch
# (great_expectations, streamlit, plotly) are pinned normally.
apache-airflow==2.9.1

pandas
numpy
psycopg2-binary
python-dotenv
pytest
pyyaml
alembic
